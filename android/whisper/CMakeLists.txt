cmake_minimum_required(VERSION 3.22)
project(LowStimSpeech LANGUAGES C CXX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Whisper and llama.cpp pin different ggml revisions; build each runtime separately.
add_subdirectory("${WHISPER_SOURCE}" upstream)
add_library(slate_speech SHARED speech.cpp)
target_compile_features(slate_speech PRIVATE cxx_std_17)
target_link_libraries(slate_speech PRIVATE whisper)
target_link_options(slate_speech PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-z,max-page-size=16384")
